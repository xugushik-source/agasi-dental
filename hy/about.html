Ընտանեկան ատամնաբուժարան՝ երկար տարիների պատմությամբ

Ագասի Կարաբեկյանի կլինիկան Ախալքալաքի առաջին ատամնաբուժարաններից մեկն է։ Այն պաշտոնապես գործում է 2003 թվականից, իսկ Ագասին 30 տարի աշխատում է որպես ատամնաբույժ։

Ընտանեկան գործ

Ագասի Կարաբեկյանը մեծացել է բժիշկների ընտանիքում և իր ամբողջ մասնագիտական կյանքը նվիրել է ատամնաբուժությանը։ Այսօր կլինիկայում նրա հետ աշխատում է նրա կինը՝ Նաիրան, որը նույնպես ատամնաբույժ է։ Երբեմն հիվանդներին ընդունում են Թբիլիսիից և Երևանից հրավիրված խորհրդատուներ։

Ամեն տարի կլինիկան ընդունում է 7–8 ուսանող-պրակտիկանտի։ Նրանք ապագա բժիշկներ են, որոնք կլինիկա են ուղարկվում ավելի փորձառու մասնագետների մոտ գործնական գիտելիքներ ձեռք բերելու համար։ Սա ինքնին խոսում է մասնագիտական հանրության շրջանում կլինիկայի և նրա մասնագետների հեղինակության մասին։

Շրջանի միակ պարոդոնտոլոգիական կաբինետը

Կլինիկան Ախալքալաքի շրջանում միակն է, որտեղ գործում է առանձին պարոդոնտոլոգիական կաբինետ՝ հագեցած պրոֆեսիոնալ մաքրման համար նախատեսված շվեյցարական EMS սարքավորումով։

Կաբինետը բացվել է Եվրոպական միության դրամաշնորհի աջակցությամբ՝ HEKS-EPER Georgia ծրագրի շրջանակում։ Սարքավորումների արժեքի 70%-ը ծածկել է դրամաշնորհը, իսկ մնացած մասը կլինիկան ֆինանսավորել է սեփական միջոցներով։

Կլինիկայում գործում է նաև առանձին վիրաբուժական կաբինետ։ Ցանկացած բարդության իմպլանտացիա իրականացնում են Թբիլիսիից և Երևանից հրավիրված լիցենզավորված բժիշկ-վիրաբույժները։ Կաբինետի սարքավորումները համապատասխանում են Վրաստանի Առողջապահության նախարարության պահանջներին։

Մոտ ապագայում նախատեսվում է նաև օրթոդոնտիկ կաբինետի բացում։
